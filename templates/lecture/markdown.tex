\newcommand{\NtwoLDeepHeading}[4]{%
  \par\Needspace{5\baselineskip}\addvspace{0.65em}%
  \phantomsection\pdfbookmark[#1]{#4}{n2l-deep-#2}%
  \def\@currentlabelname{#4}%
  \begingroup\parindent=0pt
  \ifcase\numexpr#1-3\relax\or
    \sffamily\bfseries\fontsize{11}{17}\selectfont
  \or\rmfamily\bfseries\fontsize{11}{17}\selectfont
  \else\rmfamily\fontsize{11}{17}\selectfont\fi
  #3\par\endgroup\nopagebreak[4]\vspace{0.2em}%
}

  \typeout{N2LWIDE:\NtwoLCurrentFormula}%
  \typeout{N2LDIMS:\the\wd\NtwoLMathBox:\the\linewidth}%
  \PackageError{note2lecture}{N2L_MATH_TOO_WIDE}%

\newcommand{\NtwoLDisplay}[1]{%
  \sbox{\NtwoLMathBox}{$\displaystyle #1$}%
  \ifdim\wd\NtwoLMathBox>1.12\linewidth
    \NtwoLWidthError
  \else
    \ifdim\wd\NtwoLMathBox>\linewidth
      \resizebox{\linewidth}{!}{\usebox{\NtwoLMathBox}}%
    \else
      \usebox{\NtwoLMathBox}%
    \fi
  \fi
}
